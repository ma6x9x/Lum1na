// Lum1naRoot helper v0.1 (pure C): proves the FKT lane - as a jb-packaged
// uid0 daemon, this process gets jailbreakd check-in; libkrw primitives
// should initialize here (kbase != 0). v0.2 lands the kernproc/zone
// payload. Build: xcrun -sdk iphoneos clang (see build-lum1naroot.yml).
#include <stdio.h>
#include <unistd.h>
#include <dlfcn.h>
#include <stdint.h>
#define STATIC_BASE 0xfffffff007004000ull

int main(int argc, char **argv) {
    (void)argc; (void)argv;
    void *krw = dlopen("/var/jb/usr/lib/libkrw/libkrw-dopamine.dylib", RTLD_NOW);
    if (!krw) { printf("{ \"error\": \"libkrw-dopamine dlopen failed\" }\n"); return 1; }
    uint64_t (*kbase_fn)(void) = dlsym(krw, "kbase");
    if (!kbase_fn) { printf("{ \"error\": \"kbase dlsym failed\" }\n"); return 1; }
    uint64_t kbase = kbase_fn();
    uint64_t slide = kbase ? kbase - STATIC_BASE : 0;
    printf("{ \"uid\": %d, \"kbase\": 0x%llx, \"slide\": 0x%llx, "
           "\"lane\": \"%s\" }\n", geteuid(), (unsigned long long)kbase,
           (unsigned long long)slide, kbase ? "LIVE" : "UNINITIALIZED(gSystemInfo)");
    return 0;
}
